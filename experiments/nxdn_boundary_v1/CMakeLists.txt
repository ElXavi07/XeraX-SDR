cmake_minimum_required(VERSION 3.20)
project(XeraXNxdnBoundaryV1 LANGUAGES C CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON CACHE BOOL "Retain boundary build" FORCE)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(XERAX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(XERAX_DSD "${XERAX_ROOT}/upstream/dsd-neo")
list(APPEND CMAKE_MODULE_PATH "${XERAX_DSD}/cmake")
find_package(LibSndFile REQUIRED)
find_package(mbe-neo CONFIG REQUIRED)
set(V2_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../nxdn_observation_v2")
set(V2_BUILD "${CMAKE_CURRENT_BINARY_DIR}/observation")
add_subdirectory("${V2_SOURCE}" "${V2_BUILD}")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_build.py"
    "${V2_BUILD}/generated" "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
get_target_property(BOUNDARY_PROTOCOL_SOURCES xerax_observation_protocol SOURCES)
list(FILTER BOUNDARY_PROTOCOL_SOURCES EXCLUDE REGEX "/nxdn_frame_observed\\.c$")
add_library(xerax_boundary_protocol STATIC ${BOUNDARY_PROTOCOL_SOURCES}
    "${CMAKE_CURRENT_BINARY_DIR}/generated/nxdn_frame_boundary.c")
foreach(property IN ITEMS INCLUDE_DIRECTORIES INTERFACE_INCLUDE_DIRECTORIES COMPILE_DEFINITIONS COMPILE_OPTIONS COMPILE_FEATURES LINK_LIBRARIES)
    get_target_property(value xerax_observation_protocol ${property})
    if(value)
        set_property(TARGET xerax_boundary_protocol PROPERTY ${property} "${value}")
    endif()
endforeach()
target_include_directories(xerax_boundary_protocol PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
get_target_property(ENGINE_DIR dsd-neo_engine SOURCE_DIR)
get_directory_property(ENGINE_DEFS DIRECTORY "${ENGINE_DIR}" COMPILE_DEFINITIONS)
get_target_property(ENGINE_INCLUDES dsd-neo_engine INCLUDE_DIRECTORIES)
get_target_property(CLI_LINKS dsd-neo LINK_LIBRARIES)
foreach(role IN ITEMS baseline candidate)
    if(role STREQUAL "candidate")
        set(enabled 1)
    else()
        set(enabled 0)
    endif()
    set(target "xerax_nxdn_boundary_${role}")
    add_executable(${target} phase_runtime.c phase_policy.c "${V2_SOURCE}/dispatch_observed.c"
        "${V2_BUILD}/generated/engine_observed.c" "${CMAKE_CURRENT_BINARY_DIR}/generated/observer.c"
        "${CMAKE_CURRENT_BINARY_DIR}/generated/dsd_frame_sync_boundary.c")
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${V2_SOURCE}"
        "${XERAX_DSD}/src/engine" "${XERAX_DSD}/src/dsp" "${XERAX_DSD}/src/protocol/nxdn" ${ENGINE_INCLUDES})
    target_compile_definitions(${target} PRIVATE ${ENGINE_DEFS} DSD_NEO_TEST_HOOKS
        XERAX_ACQUISITION_RESEARCH XERAX_NXDN_FRAME_RESEARCH XERAX_BOUNDARY_ENABLED=${enabled})
    target_link_libraries(${target} PRIVATE "-Wl,--start-group" xerax_boundary_protocol
        dsd-neo_dsp_private_test_support dsd-neo_engine "-Wl,--end-group"
        dsd-neo_warnings dsd-neo_feature_radio dsd-neo_feature_lz ${CLI_LINKS})
    target_link_options(${target} PRIVATE "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/${role}.map")
    foreach(function IN ITEMS getDibitSoft nxdn_voice processMbeFrameSoft processAudio rtl_stream_start rtl_stream_tune
        rtl_stream_tune_tagged rtl_stream_output_rate rtl_stream_request_fsk_reacquire)
        target_link_options(${target} PRIVATE "-Wl,--wrap=${function}")
    endforeach()
endforeach()
add_executable(xerax_nxdn_boundary_policy_test phase_policy.c test_phase_policy.c)
target_link_libraries(xerax_nxdn_boundary_policy_test PRIVATE dsd-neo_warnings)
add_test(NAME xerax_nxdn_boundary_policy_test COMMAND xerax_nxdn_boundary_policy_test)
