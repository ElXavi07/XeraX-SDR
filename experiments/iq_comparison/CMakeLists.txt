cmake_minimum_required(VERSION 3.18)
project(xerax_iq_comparison LANGUAGES CXX)
find_package(Threads REQUIRED)
set(XERAX_WHOLE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../iq_history" CACHE PATH "CreditHistory source directory; use the frozen four-file source for historical comparisons")
add_library(comparison_whole STATIC "${XERAX_WHOLE_SOURCE_DIR}/iq_history.cpp" "${XERAX_WHOLE_SOURCE_DIR}/credit_history.cpp")
target_include_directories(comparison_whole PUBLIC "${XERAX_WHOLE_SOURCE_DIR}")
add_library(comparison_slabs STATIC ../iq_slabs/immutable_slabs.cpp ../iq_coordinator/iq_coordinator.cpp)
target_include_directories(comparison_slabs PUBLIC ../iq_slabs ../iq_coordinator)
add_executable(xerax_iq_comparison comparison.cpp)
add_executable(xerax_pending_take_test test_pending_take.cpp)
target_link_libraries(xerax_pending_take_test PRIVATE comparison_slabs)
foreach(target comparison_whole comparison_slabs xerax_iq_comparison xerax_pending_take_test)
  target_compile_features(${target} PUBLIC cxx_std_17)
  target_link_libraries(${target} PUBLIC Threads::Threads)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
target_link_libraries(xerax_iq_comparison PRIVATE comparison_whole comparison_slabs)
include(CTest)
if(BUILD_TESTING)
  add_test(NAME IQ_COMPARISON_PENDING_TAKE COMMAND xerax_pending_take_test)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME IQ_COMPARISON_OBSERVER
    COMMAND "${CMAKE_COMMAND}" -E env "XERAX_COMPARISON_EXE=$<TARGET_FILE:xerax_iq_comparison>"
      "${Python3_EXECUTABLE}" -m unittest -v test_inspect_trace)
  add_test(NAME IQ_COMPARISON_OBSERVER_OPTIMIZED
    COMMAND "${CMAKE_COMMAND}" -E env "XERAX_COMPARISON_EXE=$<TARGET_FILE:xerax_iq_comparison>"
      "${Python3_EXECUTABLE}" -O -m unittest -v test_inspect_trace)
  set_tests_properties(IQ_COMPARISON_OBSERVER IQ_COMPARISON_OBSERVER_OPTIMIZED
    PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" TIMEOUT 180)
endif()
