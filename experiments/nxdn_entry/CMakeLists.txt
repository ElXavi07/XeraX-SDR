cmake_minimum_required(VERSION 3.20)
project(XeraXNxdnEntryResearch LANGUAGES C CXX)
include(CTest)
get_filename_component(XERAX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
list(APPEND CMAKE_MODULE_PATH "${XERAX_ROOT}/upstream/dsd-neo/cmake")
find_package(LibSndFile REQUIRED)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../nxdn_frames" "${CMAKE_CURRENT_BINARY_DIR}/frames")
get_target_property(OBSERVER_DIR xerax_nxdn_frame_observer SOURCE_DIR)
get_target_property(OBSERVER_SOURCES xerax_nxdn_frame_observer SOURCES)
set(SUPPORT_SOURCES "")
foreach(source IN LISTS OBSERVER_SOURCES)
    if(NOT IS_ABSOLUTE "${source}")
        set(source "${OBSERVER_DIR}/${source}")
    endif()
    if(NOT source STREQUAL "${OBSERVER_DIR}/observer.c")
        list(APPEND SUPPORT_SOURCES "${source}")
    endif()
endforeach()
get_directory_property(OBSERVER_DIR_DEFS DIRECTORY "${OBSERVER_DIR}" COMPILE_DEFINITIONS)
foreach(role IN ITEMS baseline candidate)
    add_executable(xerax_nxdn_entry_${role} observer.c ${SUPPORT_SOURCES})
    foreach(property IN ITEMS INCLUDE_DIRECTORIES COMPILE_DEFINITIONS COMPILE_OPTIONS COMPILE_FEATURES
                              C_STANDARD C_STANDARD_REQUIRED C_EXTENSIONS CXX_STANDARD CXX_STANDARD_REQUIRED CXX_EXTENSIONS
                              LINK_LIBRARIES)
        get_target_property(value xerax_nxdn_frame_observer ${property})
        if(value)
            set_property(TARGET xerax_nxdn_entry_${role} PROPERTY ${property} "${value}")
        endif()
    endforeach()
    target_compile_definitions(xerax_nxdn_entry_${role} PRIVATE ${OBSERVER_DIR_DEFS})
endforeach()
target_compile_definitions(xerax_nxdn_entry_baseline PRIVATE XERAX_ENTRY_OPTION=0)
target_compile_definitions(xerax_nxdn_entry_candidate PRIVATE XERAX_ENTRY_OPTION=1)
