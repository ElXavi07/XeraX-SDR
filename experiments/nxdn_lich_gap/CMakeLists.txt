cmake_minimum_required(VERSION 3.20)
project(XeraXNxdnLichGap LANGUAGES C CXX)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(XERAX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
list(APPEND CMAKE_MODULE_PATH "${XERAX_ROOT}/upstream/dsd-neo/cmake")
find_package(LibSndFile REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare.py" "${CMAKE_CURRENT_BINARY_DIR}/inputs" COMMAND_ERROR_IS_FATAL ANY)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../nxdn_frames" "${CMAKE_CURRENT_BINARY_DIR}/frames")
get_target_property(FRAME_SOURCES xerax_nxdn_frame_private SOURCES)
get_target_property(OBSERVER_DIR xerax_nxdn_frame_observer SOURCE_DIR)
get_directory_property(OBSERVER_DIR_DEFS DIRECTORY "${OBSERVER_DIR}" COMPILE_DEFINITIONS)
list(FILTER FRAME_SOURCES EXCLUDE REGEX "/nxdn_frame\\.c$")
add_library(xerax_nxdn_gap_private STATIC "${CMAKE_CURRENT_BINARY_DIR}/inputs/nxdn_frame_candidate.c" ${FRAME_SOURCES})
foreach(property IN ITEMS INCLUDE_DIRECTORIES INTERFACE_INCLUDE_DIRECTORIES COMPILE_DEFINITIONS COMPILE_OPTIONS COMPILE_FEATURES LINK_LIBRARIES)
    get_target_property(value xerax_nxdn_frame_private ${property})
    if(value)
        set_property(TARGET xerax_nxdn_gap_private PROPERTY ${property} "${value}")
    endif()
endforeach()
target_compile_definitions(xerax_nxdn_gap_private PRIVATE ${OBSERVER_DIR_DEFS})
foreach(role IN ITEMS baseline candidate)
    add_executable(xerax_nxdn_gap_${role} observer.c unused_dibit_support.c ../nxdn_frames/side_effect_sinks.c ../nxdn_frames/dsp_side_effect_sinks.c
        "${XERAX_ROOT}/upstream/dsd-neo/src/core/util/dsd_misc.c" "${XERAX_ROOT}/upstream/dsd-neo/src/core/util/bit_packing.c")
    foreach(property IN ITEMS INCLUDE_DIRECTORIES COMPILE_DEFINITIONS COMPILE_OPTIONS COMPILE_FEATURES LINK_LIBRARIES)
        get_target_property(value xerax_nxdn_frame_observer ${property})
        if(value)
            if(role STREQUAL "candidate" AND property STREQUAL "LINK_LIBRARIES")
                list(TRANSFORM value REPLACE "^xerax_nxdn_frame_private$" "xerax_nxdn_gap_private")
            endif()
            set_property(TARGET xerax_nxdn_gap_${role} PROPERTY ${property} "${value}")
        endif()
    endforeach()
    target_compile_definitions(xerax_nxdn_gap_${role} PRIVATE ${OBSERVER_DIR_DEFS})
endforeach()
