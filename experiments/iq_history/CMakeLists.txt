cmake_minimum_required(VERSION 3.18)
project(xerax_iq_history_experiment LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(xerax_iq_history STATIC iq_history.cpp)
target_include_directories(xerax_iq_history PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(xerax_iq_history PUBLIC cxx_std_17)
target_link_libraries(xerax_iq_history PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(xerax_iq_history PRIVATE /W4 /WX)
else()
  target_compile_options(xerax_iq_history PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(xerax_iq_history_test test.cpp)
target_link_libraries(xerax_iq_history_test PRIVATE xerax_iq_history)
add_library(xerax_iq_credit_history STATIC credit_history.cpp)
target_link_libraries(xerax_iq_credit_history PUBLIC xerax_iq_history)
add_executable(xerax_iq_credit_history_test test_credit.cpp)
target_link_libraries(xerax_iq_credit_history_test PRIVATE xerax_iq_credit_history)
if(MSVC)
  target_compile_options(xerax_iq_history_test PRIVATE /W4 /WX)
  target_compile_options(xerax_iq_credit_history PRIVATE /W4 /WX)
  target_compile_options(xerax_iq_credit_history_test PRIVATE /W4 /WX)
else()
  target_compile_options(xerax_iq_history_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(xerax_iq_credit_history PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(xerax_iq_credit_history_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_executable(xerax_iq_chunked_history_test test_chunked.cpp)
target_link_libraries(xerax_iq_chunked_history_test PRIVATE xerax_iq_credit_history)
add_executable(xerax_iq_history_latency latency.cpp)
target_link_libraries(xerax_iq_history_latency PRIVATE xerax_iq_credit_history)
target_compile_definitions(xerax_iq_history_latency PRIVATE XERAX_CHUNK_COPY=1)
add_executable(xerax_iq_observed_latency latency_observed.cpp)
target_link_libraries(xerax_iq_observed_latency PRIVATE xerax_iq_credit_history)
target_compile_definitions(xerax_iq_observed_latency PRIVATE XERAX_CHUNK_COPY=1)
foreach(check_target xerax_iq_chunked_history_test xerax_iq_history_latency xerax_iq_observed_latency)
  if(MSVC)
    target_compile_options(${check_target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${check_target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
add_test(NAME IQ_HISTORY_CONTRACT COMMAND xerax_iq_history_test)
add_test(NAME IQ_CREDIT_HISTORY_CONTRACT COMMAND xerax_iq_credit_history_test)
add_test(NAME IQ_CHUNKED_HISTORY_CONTRACT COMMAND xerax_iq_chunked_history_test)
add_test(NAME IQ_PACED_COPY_SMOKE COMMAND xerax_iq_history_latency --seconds 1 --variant chunk64 --snapshot-hz 4)
set_tests_properties(IQ_HISTORY_CONTRACT PROPERTIES TIMEOUT 30)
set_tests_properties(IQ_CREDIT_HISTORY_CONTRACT PROPERTIES TIMEOUT 30)
set_tests_properties(IQ_CHUNKED_HISTORY_CONTRACT IQ_PACED_COPY_SMOKE PROPERTIES TIMEOUT 30)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME IQ_OBSERVED_FAILURE_EVIDENCE COMMAND ${Python3_EXECUTABLE}
    -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}" -p test_observed_latency.py -v)
set_tests_properties(IQ_OBSERVED_FAILURE_EVIDENCE PROPERTIES
    ENVIRONMENT "XERAX_OBSERVED_EXE=$<TARGET_FILE:xerax_iq_observed_latency>" TIMEOUT 90)
