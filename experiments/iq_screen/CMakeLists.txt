cmake_minimum_required(VERSION 3.18)
project(xerax_iq_screen LANGUAGES CXX)
find_package(Threads REQUIRED)
set(XERAX_WHOLE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../iq_history" CACHE PATH "Exact whole-copy source directory; frozen source must be selected explicitly")
add_library(screen_whole STATIC "${XERAX_WHOLE_SOURCE_DIR}/iq_history.cpp" "${XERAX_WHOLE_SOURCE_DIR}/credit_history.cpp")
target_include_directories(screen_whole PUBLIC "${XERAX_WHOLE_SOURCE_DIR}")
add_library(screen_slabs STATIC ../iq_slabs/immutable_slabs.cpp ../iq_coordinator/iq_coordinator.cpp)
target_include_directories(screen_slabs PUBLIC ../iq_slabs ../iq_coordinator)
add_executable(xerax_iq_screen screen.cpp)
target_link_libraries(xerax_iq_screen PRIVATE screen_whole screen_slabs)
# Allocation interception is confined to this separate, untimed executable.
add_executable(xerax_iq_memory_calibration memory_calibration.cpp)
target_link_libraries(xerax_iq_memory_calibration PRIVATE screen_whole)
foreach(target screen_whole screen_slabs xerax_iq_screen xerax_iq_memory_calibration)
  target_compile_features(${target} PUBLIC cxx_std_17)
  target_link_libraries(${target} PUBLIC Threads::Threads)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  foreach(mode normal optimized)
    if(mode STREQUAL "optimized")
      set(python_flags -O)
    else()
      set(python_flags)
    endif()
    add_test(NAME IQ_SCREEN_${mode}
      COMMAND "${CMAKE_COMMAND}" -E env "XERAX_SCREEN_EXE=$<TARGET_FILE:xerax_iq_screen>"
        "XERAX_MEMORY_CALIBRATION_EXE=$<TARGET_FILE:xerax_iq_memory_calibration>"
        "${Python3_EXECUTABLE}" ${python_flags} -m unittest -v test_inspect_screen)
    add_test(NAME IQ_SCREEN_POLICY_${mode}
      COMMAND "${Python3_EXECUTABLE}" ${python_flags} -m unittest -v test_run_screen)
    set_tests_properties(IQ_SCREEN_POLICY_${mode} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" TIMEOUT 30)
    set_tests_properties(IQ_SCREEN_${mode} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" TIMEOUT 180)
  endforeach()
endif()
