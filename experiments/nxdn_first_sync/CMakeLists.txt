cmake_minimum_required(VERSION 3.20)
project(XeraXFirstNxdnSyncResearch LANGUAGES C CXX)
include(CTest)
get_filename_component(XERAX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
list(APPEND CMAKE_MODULE_PATH "${XERAX_ROOT}/upstream/dsd-neo/cmake")
find_package(LibSndFile REQUIRED)
add_subdirectory("${XERAX_ROOT}/experiments/nxdn_frames" "${CMAKE_CURRENT_BINARY_DIR}/frames")

# Preserve the ordinary private test archive as the paired baseline. Clone its
# actual source and target properties into a distinct, non-installed candidate.
set(XERAX_DSP_BASE dsd-neo_dsp_private_test_support)
get_target_property(XERAX_DSP_DIRECTORY ${XERAX_DSP_BASE} SOURCE_DIR)
get_target_property(XERAX_DSP_SOURCES ${XERAX_DSP_BASE} SOURCES)
set(XERAX_DSP_ABSOLUTE_SOURCES "")
foreach(source IN LISTS XERAX_DSP_SOURCES)
    if(IS_ABSOLUTE "${source}")
        list(APPEND XERAX_DSP_ABSOLUTE_SOURCES "${source}")
    else()
        list(APPEND XERAX_DSP_ABSOLUTE_SOURCES "${XERAX_DSP_DIRECTORY}/${source}")
    endif()
endforeach()
add_library(xerax_first_sync_candidate_dsp STATIC ${XERAX_DSP_ABSOLUTE_SOURCES})
# Per-source SIMD flags are directory-scoped too. Preserve the baseline flags
# instead of silently compiling AVX implementations with generic options.
foreach(source IN LISTS XERAX_DSP_ABSOLUTE_SOURCES)
    foreach(property IN ITEMS COMPILE_FLAGS COMPILE_OPTIONS COMPILE_DEFINITIONS)
        get_source_file_property(value "${source}" DIRECTORY "${XERAX_DSP_DIRECTORY}" ${property})
        if(value)
            set_source_files_properties("${source}" PROPERTIES ${property} "${value}")
        endif()
    endforeach()
endforeach()
foreach(property IN ITEMS INCLUDE_DIRECTORIES COMPILE_DEFINITIONS COMPILE_OPTIONS COMPILE_FEATURES LINK_LIBRARIES
                          C_STANDARD C_STANDARD_REQUIRED C_EXTENSIONS CXX_STANDARD CXX_STANDARD_REQUIRED CXX_EXTENSIONS)
    get_target_property(value ${XERAX_DSP_BASE} ${property})
    if(value)
        set_property(TARGET xerax_first_sync_candidate_dsp PROPERTY ${property} "${value}")
    endif()
endforeach()
get_directory_property(XERAX_DSP_DIRECTORY_DEFINITIONS DIRECTORY "${XERAX_DSP_DIRECTORY}" COMPILE_DEFINITIONS)
target_compile_definitions(xerax_first_sync_candidate_dsp PRIVATE ${XERAX_DSP_DIRECTORY_DEFINITIONS})
target_compile_definitions(xerax_first_sync_candidate_dsp PRIVATE XERAX_FIRST_CANONICAL_NXDN48)

# Reuse exact observer/sink sources, replacing only its private DSP link.
get_target_property(XERAX_OBSERVER_DIRECTORY xerax_nxdn_frame_observer SOURCE_DIR)
get_target_property(XERAX_OBSERVER_SOURCES xerax_nxdn_frame_observer SOURCES)
set(XERAX_OBSERVER_ABSOLUTE_SOURCES "")
foreach(source IN LISTS XERAX_OBSERVER_SOURCES)
    if(IS_ABSOLUTE "${source}")
        list(APPEND XERAX_OBSERVER_ABSOLUTE_SOURCES "${source}")
    else()
        list(APPEND XERAX_OBSERVER_ABSOLUTE_SOURCES "${XERAX_OBSERVER_DIRECTORY}/${source}")
    endif()
endforeach()
add_executable(xerax_nxdn_first_sync_candidate ${XERAX_OBSERVER_ABSOLUTE_SOURCES})
foreach(property IN ITEMS INCLUDE_DIRECTORIES COMPILE_DEFINITIONS COMPILE_OPTIONS COMPILE_FEATURES
                          C_STANDARD C_STANDARD_REQUIRED C_EXTENSIONS CXX_STANDARD CXX_STANDARD_REQUIRED CXX_EXTENSIONS)
    get_target_property(value xerax_nxdn_frame_observer ${property})
    if(value)
        set_property(TARGET xerax_nxdn_first_sync_candidate PROPERTY ${property} "${value}")
    endif()
endforeach()
get_directory_property(XERAX_OBSERVER_DIRECTORY_DEFINITIONS DIRECTORY "${XERAX_OBSERVER_DIRECTORY}" COMPILE_DEFINITIONS)
target_compile_definitions(xerax_nxdn_first_sync_candidate PRIVATE ${XERAX_OBSERVER_DIRECTORY_DEFINITIONS})
get_target_property(XERAX_OBSERVER_LINKS xerax_nxdn_frame_observer LINK_LIBRARIES)
list(TRANSFORM XERAX_OBSERVER_LINKS REPLACE "^dsd-neo_dsp_private_test_support$" "xerax_first_sync_candidate_dsp")
target_link_libraries(xerax_nxdn_first_sync_candidate PRIVATE ${XERAX_OBSERVER_LINKS})
