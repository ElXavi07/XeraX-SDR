cmake_minimum_required(VERSION 3.18)
project(xerax_immutable_slabs_experiment LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(xerax_immutable_slabs STATIC immutable_slabs.cpp)
target_include_directories(xerax_immutable_slabs PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(xerax_immutable_slabs PUBLIC cxx_std_17)
target_link_libraries(xerax_immutable_slabs PUBLIC Threads::Threads)
add_executable(xerax_immutable_slabs_test test_immutable_slabs.cpp)
target_link_libraries(xerax_immutable_slabs_test PRIVATE xerax_immutable_slabs)
foreach(check_target xerax_immutable_slabs xerax_immutable_slabs_test)
  if(MSVC)
    target_compile_options(${check_target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${check_target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
enable_testing()
add_test(NAME IQ_IMMUTABLE_SLABS_CONTRACT COMMAND xerax_immutable_slabs_test)
set_tests_properties(IQ_IMMUTABLE_SLABS_CONTRACT PROPERTIES TIMEOUT 60)
