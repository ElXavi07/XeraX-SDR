cmake_minimum_required(VERSION 3.18)
project(xerax_iq_notification LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(xerax_ready_notification STATIC ready_notification.cpp)
target_include_directories(xerax_ready_notification PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(xerax_ready_notification PUBLIC cxx_std_17)
target_link_libraries(xerax_ready_notification PUBLIC Threads::Threads)

include(CTest)
set(notification_targets xerax_ready_notification)
if(BUILD_TESTING)
  # Existing source ownership cores are linked unchanged for one integration
  # correctness check. No observer, benchmark or application is changed.
  add_library(notification_source STATIC ../iq_slabs/immutable_slabs.cpp ../iq_coordinator/iq_coordinator.cpp)
  target_include_directories(notification_source PUBLIC ../iq_slabs ../iq_coordinator)
  target_compile_features(notification_source PUBLIC cxx_std_17)
  target_link_libraries(notification_source PUBLIC Threads::Threads)
  add_executable(xerax_ready_notification_test test_ready_notification.cpp)
  target_link_libraries(xerax_ready_notification_test PRIVATE xerax_ready_notification notification_source)
  # Explicit checks in the test must remain effective with ordinary assert off.
  target_compile_definitions(xerax_ready_notification_test PRIVATE NDEBUG)
  list(APPEND notification_targets notification_source xerax_ready_notification_test)
  add_test(NAME IQ_READY_NOTIFICATION_CONTRACT COMMAND xerax_ready_notification_test)
  set_tests_properties(IQ_READY_NOTIFICATION_CONTRACT PROPERTIES TIMEOUT 120)
  add_executable(xerax_iq_notification_mailbox_test test_mailbox_notification.cpp)
  target_link_libraries(xerax_iq_notification_mailbox_test PRIVATE xerax_ready_notification notification_source)
  target_compile_definitions(xerax_iq_notification_mailbox_test PRIVATE NDEBUG)
  list(APPEND notification_targets xerax_iq_notification_mailbox_test)
  add_test(NAME IQ_NOTIFICATION_MAILBOX_CONTRACT COMMAND xerax_iq_notification_mailbox_test)
  set_tests_properties(IQ_NOTIFICATION_MAILBOX_CONTRACT PROPERTIES TIMEOUT 60)
endif()
foreach(target IN LISTS notification_targets)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
