cmake_minimum_required(VERSION 3.18)
project(xerax_iq_coordinator_experiment LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(xerax_coordinator_slabs STATIC ../iq_slabs/immutable_slabs.cpp)
target_include_directories(xerax_coordinator_slabs PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../iq_slabs")
target_compile_features(xerax_coordinator_slabs PUBLIC cxx_std_17)
target_link_libraries(xerax_coordinator_slabs PUBLIC Threads::Threads)
add_library(xerax_iq_coordinator STATIC iq_coordinator.cpp)
target_include_directories(xerax_iq_coordinator PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(xerax_iq_coordinator PUBLIC xerax_coordinator_slabs)
add_executable(xerax_iq_coordinator_test test_coordinator.cpp)
target_link_libraries(xerax_iq_coordinator_test PRIVATE xerax_iq_coordinator)
foreach(check_target xerax_coordinator_slabs xerax_iq_coordinator xerax_iq_coordinator_test)
  if(MSVC)
    target_compile_options(${check_target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${check_target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
enable_testing()
add_test(NAME IQ_COORDINATOR_CONTRACT COMMAND xerax_iq_coordinator_test)
set_tests_properties(IQ_COORDINATOR_CONTRACT PROPERTIES TIMEOUT 90)
