cmake_minimum_required(VERSION 3.20)
project(XeraXIndependentVoiceWords LANGUAGES C CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON CACHE BOOL "Preserve build identity" FORCE)
find_package(mbe-neo CONFIG REQUIRED)
if(NOT VOICE_PRIMARY_ROOT)
    message(FATAL_ERROR "Set VOICE_PRIMARY_ROOT to the verified pinned source closure")
endif()
get_filename_component(XERAX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(MMDVM "${VOICE_PRIMARY_ROOT}/MMDVM-Host")
add_executable(xerax_voice_primary primary_encoder.cpp primary_support.cpp
    "${MMDVM}/NXDNAudio.cpp" "${MMDVM}/Golay24128.cpp")
target_include_directories(xerax_voice_primary PRIVATE "${MMDVM}")
target_compile_features(xerax_voice_primary PRIVATE cxx_std_17)
add_executable(xerax_voice_probe probe.c)
target_include_directories(xerax_voice_probe PRIVATE "${XERAX_ROOT}/upstream/dsd-neo/include")
target_link_libraries(xerax_voice_probe PRIVATE mbe_neo::mbe_static)
if(MINGW)
    # The existing mbelib archive uses MinGW's emulated thread-local storage.
    target_link_libraries(xerax_voice_probe PRIVATE pthread)
endif()
target_compile_features(xerax_voice_probe PRIVATE c_std_11)
foreach(target IN ITEMS xerax_voice_primary xerax_voice_probe)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    target_link_options(${target} PRIVATE "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/${target}.map")
endforeach()
