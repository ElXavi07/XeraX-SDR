cmake_minimum_required(VERSION 3.18)
project(xerax_iq_notify_screen LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(notify_whole STATIC ../iq_history/iq_history.cpp ../iq_history/credit_history.cpp)
target_include_directories(notify_whole PUBLIC ../iq_history)
add_library(notify_source STATIC ../iq_slabs/immutable_slabs.cpp ../iq_coordinator/iq_coordinator.cpp ../iq_notification/ready_notification.cpp)
target_include_directories(notify_source PUBLIC ../iq_slabs ../iq_coordinator ../iq_notification)
add_executable(xerax_iq_notify_screen notify_screen.cpp)
target_link_libraries(xerax_iq_notify_screen PRIVATE notify_whole notify_source)
foreach(target notify_whole notify_source xerax_iq_notify_screen)
  target_compile_features(${target} PUBLIC cxx_std_17)
  target_link_libraries(${target} PUBLIC Threads::Threads)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  foreach(mode normal optimized)
    if(mode STREQUAL "optimized")
      set(python_flags -O)
    else()
      set(python_flags)
    endif()
    add_test(NAME IQ_NOTIFY_SCREEN_${mode}
      COMMAND "${CMAKE_COMMAND}" -E env "XERAX_NOTIFY_SCREEN_EXE=$<TARGET_FILE:xerax_iq_notify_screen>"
        "${Python3_EXECUTABLE}" ${python_flags} -m unittest -v test_inspect_notify)
    set_tests_properties(IQ_NOTIFY_SCREEN_${mode} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" TIMEOUT 180)
    add_test(NAME IQ_NOTIFY_SCREEN_POLICY_${mode}
      COMMAND "${Python3_EXECUTABLE}" ${python_flags} -m unittest -v test_run_notify_screen)
    set_tests_properties(IQ_NOTIFY_SCREEN_POLICY_${mode} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" TIMEOUT 30)
  endforeach()
endif()
